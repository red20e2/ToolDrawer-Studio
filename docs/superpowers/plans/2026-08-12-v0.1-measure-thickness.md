# ToolDrawer Studio V0.1 Measure / Thickness Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a calibrated side-view Measure stage that estimates a tool's maximum physical thickness, supports manual correction and exact override, derives safe pocket-depth suggestions, preserves explicit manufacturing overrides, and migrates existing `.tds` projects to schema V2 without changing old pocket geometry.

**Architecture:** Add a focused `measurement` package containing immutable measurement result types, depth-resolution rules, and deterministic OpenCV thickness analysis. Keep project mutation and stale-state invalidation in `WorkflowController`; keep Qt widgets presentation-only and route all manufacturing dimensions through the controller. Persist source measurement state in `.tds` schema V2 with an explicit V1-to-V2 migration, and treat this branch as a stacked feature on top of `feature/v0.1-capture-workflow` until PR #3 is explicitly merged.

**Tech Stack:** Python 3.12, PySide6/Qt 6, NumPy 2.x, OpenCV 4.x, CadQuery 2.x, pytest 8.x, GitHub Actions Windows runner.

## Global Constraints

- Branch: `feature/v0.1-measure-thickness`, based on capture-workflow head `9141e5107b2899d4d2bc48617424acbd50690063`.
- Do not merge or retarget PR #3 as part of this implementation.

- Internal geometry units remain millimetres.
- One active side-view capture per tool in V0.1.
- Every side-view image requires its own calibration record; never reuse another capture's scale by assumption.
- Automatic thickness uses deterministic OpenCV/image processing only; no AI-model dependency is added in this slice.
- High-confidence automatic thickness is accepted at confidence `>= 0.80`; lower confidence requires explicit acceptance, endpoint correction, or exact manual thickness.
- Physical thickness and manufacturing pocket depth are separate values.
- Project defaults are exposed height `4.0 mm` and bottom clearance `0.8 mm`.
- Suggested depth formula is `accepted_thickness_mm - effective_exposed_height_mm + effective_bottom_clearance_mm`.
- Explicit final pocket-depth overrides are never silently replaced by remeasurement.
- V1 `.tds` projects migrate to V2 non-destructively; old `depth_mm` becomes `pocket_depth_override_mm` exactly.
- Replacing/recalibrating a side view invalidates image-derived automatic/endpoint measurements, preserves exact manual thickness, preserves explicit pocket override, and marks preserved values for review.
- Selecting a pending capture as a side view uses the existing non-consuming promotion path; the pending item remains in the tray.
- Manual two-point measurement must work even when automatic silhouette detection fails.
- Unsupported future `.tds` schema versions remain rejected.
- Full existing test suite must stay green; final verification must run on Windows CI.

Create `measurement/depth.py`:

```python
from __future__ import annotations

from math import isfinite

from tooldrawer_studio.domain.models import Project, ToolObject


def _nonnegative(value: float, label: str) -> float:
    value = float(value)
    if not isfinite(value) or value < 0:
        raise ValueError(f"{label} must be finite and non-negative")
    return value


def _positive(value: float, label: str) -> float:
    value = float(value)
    if not isfinite(value) or value <= 0:
        raise ValueError(f"{label} must be finite and positive")
    return value


def effective_exposed_height_mm(project: Project, tool: ToolObject) -> float:
    value = (
        project.default_exposed_height_mm
        if tool.exposed_height_override_mm is None
        else tool.exposed_height_override_mm
    )
    return _nonnegative(value, "Exposed height")


def effective_bottom_clearance_mm(project: Project, tool: ToolObject) -> float:
    value = (
        project.default_bottom_clearance_mm
        if tool.bottom_clearance_override_mm is None
        else tool.bottom_clearance_override_mm
    )
    return _nonnegative(value, "Bottom clearance")


def suggested_pocket_depth_mm(project: Project, tool: ToolObject) -> float | None:
    if not tool.thickness_accepted or tool.accepted_thickness_mm is None:
        return None
    thickness = _positive(tool.accepted_thickness_mm, "Accepted thickness")
    depth = (
        thickness
        - effective_exposed_height_mm(project, tool)
        + effective_bottom_clearance_mm(project, tool)
    )
    if not isfinite(depth) or depth <= 0:
        raise ValueError("Measurement settings must produce a positive pocket depth")
    return float(depth)


def final_pocket_depth_mm(project: Project, tool: ToolObject) -> float | None:
    if tool.pocket_depth_override_mm is not None:
        return _positive(tool.pocket_depth_override_mm, "Pocket depth override")
    return suggested_pocket_depth_mm(project, tool)
```

Create `persistence/migrations.py`:

```python
from __future__ import annotations

from copy import deepcopy

CURRENT_SCHEMA_VERSION = 2


def _migrate_v1_to_v2(data: dict) -> dict:
    migrated = deepcopy(data)
    migrated["schema_version"] = 2
    migrated.setdefault("default_exposed_height_mm", 4.0)
    migrated.setdefault("default_bottom_clearance_mm", 0.8)
    for tool in migrated.get("tools", []):
        old_depth = float(tool.pop("depth_mm", 5.0))
        tool.setdefault("side_view_capture_id", None)
        tool.setdefault("automatic_thickness_mm", None)
        tool.setdefault("automatic_thickness_confidence", None)
        tool.setdefault("automatic_thickness_endpoint_a_px", None)
        tool.setdefault("automatic_thickness_endpoint_b_px", None)
        tool.setdefault("corrected_thickness_endpoint_a_px", None)
        tool.setdefault("corrected_thickness_endpoint_b_px", None)
        tool.setdefault("side_view_silhouette_px", [])
        tool.setdefault("accepted_thickness_mm", None)
        tool.setdefault("thickness_measurement_mode", "none")
        tool.setdefault("thickness_accepted", False)
        tool.setdefault("exposed_height_override_mm", None)
        tool.setdefault("bottom_clearance_override_mm", None)
        tool.setdefault("pocket_depth_override_mm", old_depth)
        tool.setdefault("thickness_review_required", False)
    return migrated


def migrate_project_dict(data: dict, from_version: int) -> dict:
    if from_version < 1 or from_version > CURRENT_SCHEMA_VERSION:
        raise ValueError(f"Unsupported project schema version: {from_version}")
    migrated = deepcopy(data)
    version = from_version
    while version < CURRENT_SCHEMA_VERSION:
        if version == 1:
            migrated = _migrate_v1_to_v2(migrated)
            version = 2
            continue
        raise ValueError(f"Unsupported project schema version: {version}")
    migrated["schema_version"] = CURRENT_SCHEMA_VERSION
    return migrated
```

def _threshold_masks(pixels_bgr: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    gray = cv2.cvtColor(pixels_bgr, cv2.COLOR_BGR2GRAY)
    blurred = cv2.GaussianBlur(gray, (5, 5), 0)
    _, light = cv2.threshold(blurred, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)
    return light, cv2.bitwise_not(light)
```

mm = _map_pixels_to_mm(contour_xy_px, calibration)

long_axis = vh[0]
normal_axis = np.array([-long_axis[1], long_axis[0]])
long_pos = centered @ long_axis
normal_pos = centered @ normal_axis
```

confidence = float(calibration.confidence)
